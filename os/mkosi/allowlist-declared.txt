# The files a VM can't open, kept in the image with the traced ones (os/PLAN.md, OS-4;
# os/mkosi/prune.py). Globs: "**" crosses folders, "*" and "?" don't. A line starting
# with "debug:" is for the debug image only. Kernel modules and firmware aren't listed:
# mkosi.conf's KernelModules= chooses them by hardware class, and mkosi keeps the
# firmware they ask for. /etc is the image's configuration and is kept whole.

# Read by systemd before the recorder starts, or run after it stops: units, generators,
# presets, the settings early services read, and the shutdown. No per-user manager runs
# (user@.service is masked), so none of its units or generators are kept.
/usr/lib/os-release
/usr/lib/systemd/system/**
# Generators run on every boot; the stick needs only its fstab and the kernel command
# line's masks. gpt-auto would mount partitions by type, which the stick never wants.

/usr/lib/systemd/system-generators/systemd-debug-generator
/usr/lib/systemd/system-preset/**
/usr/lib/systemd/system-shutdown/**
/usr/lib/systemd/system-sleep/**
/usr/lib/systemd/systemd
/usr/lib/systemd/systemd-executor
/usr/lib/systemd/systemd-shutdown
/usr/lib/systemd/systemd-sleep
/usr/lib/systemd/*.conf

/usr/lib/tmpfiles.d/**

/usr/share/dbus-1/**
# udev's rules, its helpers and the hardware database: they name and set up hardware
# the VM doesn't have (a laptop's keys, its backlight, its touchpad).
/usr/lib/udev/**
/usr/share/libinput/**

/usr/share/fonts/**
/usr/share/fontconfig/**

/usr/share/glvnd/**
/usr/share/drirc.d/**
/usr/share/libdrm/**
# The C locale. CPU microcode: none is installed (the firmware's own is used).
/usr/lib/locale/**
